        \subsubsection*{Dijkstra (Link-State)}

            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{Centralizado:} Todos os roteadores sabem os custos dos enlaces na rede
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Difusão do estado do estado de enlace, a informação é propagada por toda a rede via \textit{broadcast}
                    \item Todos os nós possuem a mesma informação sobre o estado da rede 
                \end{itemize}
                \item Ideia: Computa o caminho de menor custo de um nó (origem) para todos os outros nós
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Cada roteador executa o algoritmo de Dijkstra, calculando os caminhos de menor custo dele para os demais nós da rede
                    \item Ao fim desse cálculo, o roteador alimenta sua \textbf{tabela de encaminhamento} para esse nó  
                \end{itemize}
                \item \textbf{Iterativo:} Após $k$ iterações, o nó conhece os caminhos de menor custo para $k$ destinos  
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item A cada iteração obtem-se a informação de alcançabilidade, caminho de menor custo, para um determinado nó  
                \end{itemize}   
                \item \textbf{Notação:} 
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item $\mathbf{c_{x,y}}:$ Custo \underline{direto} do enlace entre os nós $x$ e $y$, $= \infty$ se não houver enlace
                    \item $\mathbf{D(v)}:$ A atual \underline{estimativa} do custo do caminho de menor custo do nó de origem até o nó $v$
                    \item $\mathbf{p(v)}:$ Predecessor, nó que vem antes, de $v$ no caminho de menor custo do nó de origem até $v$
                    \item $\mathbf{N'}:$ Conjuntos de nós para os quais o algoritmo já sabe \textit{definitivamente} o caminho de menor custo
                \end{itemize}
            \end{itemize}
        
            \begin{algorithm}
                \caption{Algoritmo de Dijkstra}
                \begin{algorithmic}[1]
                    \State \textbf{Initialization:}
                    \State $N' = \{u\}$  /* Computa o caminho de menor custo de $u$ para os outros nós */
                    \For{all nodes $v$}
                        \If{$v$ adjacent to $u$}  /* $u$ conhece só o custo direto apenas para vizinhos*/
                            \State $D(v) = c_{u,v}$  /* Mas, pode não ser o custo mínimo! */
                        \Else
                            \State $D(v) = \infty$
                        \EndIf
                    \EndFor

                    \State
                    \State \textbf{Loop:}
                    \Repeat
                        \State find $w$ not in $N'$ such that $D(w)$ is minimum
                        \State add $w$ to $N'$
                        \State update $D(v)$ for all $v$ adjacent to $w$ and not in $N'$:
                        \State $D(v) = \min(D(v), D(w) + c_{w,v})$
                        \State /* O novo caminho mínimo para $v$ é o antigo custo de $v$ ou
                        \State \ \ o custo mínimo conhecido para $w$ + o custo direto de $w$ para $v$. */
                    \Until{all nodes in $N'$}
                
                \end{algorithmic}
            \end{algorithm}

            \textbf{Complexidade do Algoritmo :} $n$ nós.
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item Cada uma das $n$ iterações : Precisa chegar todos os nós,$w$, que não estão em $N'$.
                \item $\frac{n(n+1)}{2}$ comparações : Complexidade $O(n^2)$
                \item Implementação mais eficiente possível : Complexidade $O(nlog n)$ 
            \end{itemize}

            \textbf{Complexidade da Mensagem :}
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item Cada roteador tem que fazer \textbf{broadcast}(difusão de informção) do estado de enlace para todos os outros $n$ roteadores.
                \item Existem algoritmos eficientes de broadcast: $O(n)$ cruzamentos de enlaces para disseminar uma mensagem broadcast de uma origem.
                \item Cada roteador tem que divulgar $O(n)$ mensagens: Complexidade de mensagem final $O(n^2)$.
            \end{itemize}

        \subsubsection*{Equação de \textbf{Bellman-Ford} (Distance-Vector)}

            \textbf{Equação de Bellman-Ford:}
            \begin{align*}
                D_x(y) : \text{Custo do caminhho de menor custo de $x$ para $y$} \\
                D_x(y) = \min_{v} \{c_{x,v} + D_v(y)\} \\
                c_{x,v} : \text{Custo direto do enlace de $x$ para $v$}
            \end{align*}    
        
           \textit{Exemplo :} Suponha que os nós vizinhos de $u$, $x,y,w$ sabem o destino para $z$ 

            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/bellman-ford-exemplo.png}
            \end{center}
            
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{Ideia Principal :}
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item De tempos em tempos, cada nó \textbf{envia} seu própria estimativa do vetor de distâncias para todos os seus vizinhos.
                    \item Quando um nó $x$ recebe uma estimativa do vetor de distâncias ($DV$) de um vizinho, ele \textbf{atualiza} sua própria 
                    estimativa do vetor de distâncias, usando a equação de \textit{Bellman-Ford}.
                    \begin{align*}
                        D_x(y) \leftarrow \min_{v} \{c_{x,v} + D_v(y)\} \text{ para cada nó } y \in N
                    \end{align*} 
                    \item Em codições normais, a estimativa $D_x(y)$ converge para o atual menor custo de $x$, $d_x(y)$.
                \end{itemize}
                
                \item \textbf{Para cada nó :}
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item \textbf{Espera} por uma mudança no custo de um enlace local ou por uma mensagem de atualização de um vizinho.
                    \item \textbf{Recalcula} suas estimativas do vetor de distâncias $(DV)$ baseado nas novas informações recebidas do vizinho.
                    \item \textbf{Se} o vetor de distâncias tiver alguma modificação, \textbf{notifica} os vizinhos.
                    \item O processo se repete.
                \end{itemize}
            
                \item \textbf{Iterativo, Assíncrono :} Cada iteração local é causado a por
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Mudança no custo do enlace local
                    \item Mensagem de atualização do $DV$ de um vizinho
                \end{itemize}
            
                \item  \textbf{Distribuído, Auto-controlado :} Cada nó notifica seus vizinhos somente quando seu $DV$ muda. 
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Vizinhos então notificam seus vizinhos, somente se necessário.
                    \item Sem notificações recebidas, sem ações tomadas.
                \end{itemize}
            
            \end{itemize}

        \subsubsection*{Comparação entre Estado de Enlace e Vetor de Distância}

            \textbf{Complexidade da mensagem} 
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{LS:} \textbf{n} roteadores, $O(n^2)$ mensagens enviadas 
                \item \textbf{DV:} Intercambio de informação entre vizinhos, tempo de convergência varia
            \end{itemize}

            \textbf{Velocidade de Convergência} 
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{LS:} Algoritmo $O(n^2)$, mensagens $O(n^2)$.   
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Pode ter oscilações
                \end{itemize}
                \item \textbf{DV:} Tempo de convergência varia
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Pode ter loops de roteamento
                    \item Contagem ao infinito
                \end{itemize}
            \end{itemize}

            \textbf{Robustez:} O que acontece se um roteador tiver um mal funcionamento ou ser comprometido?
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{LS:}  
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Roteador pode fazer a divulgação errada do custo do \textbf{enlace}
                    \item Cada roteador computa somente a sua própria tabela de encaminhamento
                \end{itemize}
                \item \textbf{DV:}
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Roteador pode fazer a divulgação incorreta do custo do \textbf{caminho} : Buraco negro 
                    \item Cada tabela do roteador é usado pelos outros : erro é propagado pela rede.
                \end{itemize}
            \end{itemize}

    \subsection{Roteamento intra-ISP : OSPF}

        \subsubsection*{Escalabilidade}
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item Até agora nos estudos de roteamento assumimos um cenário idealizado
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Todos os roteadores são idênticos
                \item Rede é \textit{flat}, sem hierarquia
            \end{itemize}
            \item O que não ocorre na prática 
            
            \item \textbf{Escala:} bilhões de destinos
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Impossível armazenar todos os destinos nas tabelas de encaminhamento!
                \item ``Inundando'' a internet de tanta mensagem de controle
            \end{itemize}
            \item \textbf{Autonomia Administrativa:} 
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Internet: Conjunto de redes interligadas
                \item Cada administrador da rede quer controlar o roteamento da sua própria rede.
            \end{itemize}

            \item \textbf{Solução:} Agregar roteadores em \textbf{regiões} conhecidas como \textbf{Sistemas Autônomos (AS)}
            
            \item \textbf{intra-AS(intra-domínio):} Roteamento entre os roteadores de uma mesma AS    
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Todos os roteadores dentro de uma mesma AS executam o mesmo protocolo de roteamento \underline{intra-domínio}.
                \item Roteadores em AS diferentes podem executar protocolos de roteamento intra-domínios diferentes.
                \item \textbf{Roteador \textit{gateway}:} Roteador que fica na borda de rede conectando uma AS a outra AS.
            \end{itemize}
            
            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/intra-as.png}
            \end{center}

            \item \textbf{inter-AS(inter-domínio):} Roteamento \textbf{entre} ASs  
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Os \textit{gateways} realizam roteamento entre domínios. 
            \end{itemize}

        \subsubsection*{Roteamento Inter-AS} : A função do intra-domínio para o encaminhamento
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item Suponha que o roteador em $AS1$ receba um datagrama destinado para fora do $AS1$
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item O roteador deveria encaminhar o pacote para um roteador \textit{gateway}, mas qual ?
            \end{itemize}
            \item \textbf{O roteamento inter-domínio AS1 deve:} 
            \begin{enumerate}[left=0.5cm, align=left, nosep]
                \item Aprender quais destinos são atingíveis através dos \textit{gateways} AS2, AS3
                \item Propagar essa informação de alcançabilidade para todos os roteadores dentro do AS1  
            \end{enumerate}
        \end{itemize}

        \begin{center}
            \includegraphics[width=0.5\textwidth]{img/cap-04.01/inter-domain.png}
        \end{center}

        \subsubsection*{Roteamento Intra-AS}
            Protocolos de roteamento intra-domínio mais comuns:
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item \textbf{RIP: Routing Information Protocol} 
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Clássico protocolo de \underline{vetor de distância}: DVs trocam a cada 30 segundos
                    \item Não é muito usado hoje em dia 
                \end{itemize}
            
                \item \textbf{EIGRP: Enhanced Interior Gateway Routing Protocol} 
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Baseado em \underline{vetor de distância}
                    \item Originalmente desenvolvido pela \textit{Cisco}, hoje um padrão aberto
                \end{itemize}

                \item \textbf{OSPF: Open Shortest Path First} 
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item Roteamento de \underline{Estado de enlace}
                    \item protocolo IS-IS, essencialmente o mesmo que OSPF
                    \item Mais utilizado atualmente
                \end{itemize}
            
            \end{itemize}

        \subsubsection*{Hierarquia OSPF}
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item \textbf{Dois níveis hierárquicos :} \textit{Áreas Locais} e \textit{Backbone}
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Minimizar o impacto da ``Inundação'' nas redes
                \item O estado de enlace anuncia a inundação somente na Area ou no Backbone
                \item Cada nó tem detalhadamente a área topologica, sabem somente a direção para alcançar outros destinos
            \end{itemize}
        
            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/hierarquia-ospf.png}
            \end{center}

        \end{itemize}

    \subsection{Roteamento entre ISPs: BGP}
    
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item É realizado por um único protocolo
            \item \textbf{BGP: Border Gateway Protocol}
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item A ``cola'' que mantém a internet unida.
            \end{itemize}
            \item Permite que uma sub-rede anuncie sua existência e os destinos que pode alcançar para o resto da internet.
            \item Para cada AS o BGP provê:  
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item \textbf{eBGP:} Obtém informações sobre quais as subredes que são alcançáveis a partir dos ASes vizinhos 
                \item \textbf{iBGP:} Propaga a informação de alcançabilidade para todos os roteadores dentro do AS.
                \item Determina ``boas'' rotas para outras redes baseadas em informações e \textbf{políticas} de alcance.
            \end{itemize}   
        
            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/conexoes-bgps.png}
            \end{center}

        \end{itemize}
    
        \subsubsection*{Funcionamento Básico}
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item \textbf{Sessão BGP:} Dois roteadores BGP (``pares'') trocam mensagens BGP através de uma conexão semi-permanente TCP:   
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Anunciam \textbf{caminhos} para diferentes destinos de prefixos na rede
                \item Protocolo de ``vetor de caminhos'', rota de sistemas autônomos 
            \end{itemize}
            
            \item \textit{Exemplo:} Quando um gateway 3a AS3 anunciam o caminho AS3,X para o gateway 2c em AS2:  
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item O AS3 \textbf{divulga} ao AS2 que é possível alcançar/enviar datagramas para X por mim. 
            \end{itemize}

            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/exemplo-bgp.png}
            \end{center}

            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                 \item O roteador 2c do AS2 recebe o anúncio de caminho \textbf{AS3, X} (via eBGP) do roteador 3a do AS3.
                \item Com base na política do AS2, o roteador 2c do AS2 aceita o caminho \textbf{AS3, X} e o propaga (via iBGP) para todos os roteadores do AS2.
                \item Com base na política do AS2, o roteador 2a do AS2 anuncia (via eBGP) o caminho \textbf{AS2, AS3, X} para o roteador 1c do AS1.
            \end{itemize}
        
            \item \textit{Exemplo2:} Múltiplos caminhos

            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/exemplo2-bgp.png}
            \end{center}

            \item O roteador gateway pode aprender sobre \textbf{múltiplos} caminhos até o destino:
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item O roteador gateway do AS1, 1c, aprende o caminho \textbf{AS2, AS3, X} a partir de 2a.
                \item O roteador gateway do AS1, 1c, aprende o caminho \textbf{AS3, X} a partir de 3a.
                \item Com base na \textit{política}, o roteador gateway 1c do AS1 escolhe o caminho \textbf{AS3, X} e anuncia esse caminho dentro do AS1 via iBGP.
            \end{itemize}
            
        \end{itemize}

        \subsubsection*{Mensagem BGP}
    
            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item Mensagens BGP são trocadas entre pares através de uma conexão TCP
                \item Tipos:
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item \textbf{OPEN:} abre uma conexão TCP com o par BGP remoto e autentica o par BGP remetente.  
                    \item \textbf{UPDATE:} anuncia um novo caminho (ou retira um antigo).
                    \item \textbf{KEEPALIVE:} mantém a conexão ativa na ausência de mensagens UPDATE; também confirma (ACK) o pedido OPEN.
                    \item \textbf{NOTIFICATION:} reporta erros em mensagens anteriores; também é usada para encerrar a conexão.
                \end{itemize}
            
                \item \textit{Voltando ao Exemplo2:}
                
                \begin{center}
                    \includegraphics[width=0.15\textwidth]{img/cap-04.01/caminho-bgp.png}
                \end{center}
                
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item lembrar: 1a, 1b, 1d aprendem via iBGP a partir de 1c: ``o caminho para X passa por 1c''
                    \item em 1d: roteamento intra-domínio OSPF: para chegar a 1c, use a interface 1
                    \item em 1d: para chegar a X, use a interface 1
                    \item em 1a: roteamento intra-domínio OSPF: para chegar a 1c, use a interface 2 
                \end{itemize}

                \begin{center}
                    \includegraphics[width=0.15\textwidth]{img/cap-04.01/caminho-bgp2.png}
                \end{center}

                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item lembrar: 1a, 1b, 1d aprendem via iBGP a partir de 1c: ``o caminho para X passa por 1c''
                    \item em 1d: roteamento intra-domínio OSPF: para chegar a 1c, use a interface 1
                    \item em 1d: para chegar a X, use a interface 1 
                \end{itemize}
                
            \end{itemize}
    
        \subsubsection*{Difirença entre Intra-, Inter-AS}
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item \textbf{Políticas :}
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item \underline{inter-AS}: Admin quer ter o controle por onde o trafego será roteado, ou quem vai poder rotear o trafego através do seu AS. 
                \item \underline{intra-AS}: Admin única, questões políticas não são tão relevantes.
            \end{itemize}
            
            \item \textbf{Escalabilidade :}  
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Roteamento hierarquico salva bastante espaço nas tabelas de encaminhamento, reduzir o trafego de atualizações.
            \end{itemize}
        
            \item \textbf{Desempenho :}
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item \underline{inter-AS}: política domina, é mais relevante que o desempenho. 
                \item \underline{intra-AS}: Pode focar no desempenho.
            \end{itemize}

        \end{itemize}
        
        \subsubsection*{Roteamento Hot Potato}
        
            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/roteamento-hot-potato.png}
            \end{center}

            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item 2d aprende (via iBGP) que pode rotear para X via 2a ou 2c
                \item Roteamento ``batata quente'' (\textit{hot potato}): escolher o gateway local que tem menor custo \textit{intra-domínio}
                \item \textit{Exemplo:} 2d escolhe 2a, mesmo tendo mais saltos de AS até X; não se preocupar com o custo \textit{inter-domínio}!
            \end{itemize}
        
        \subsubsection*{Definição das política}

            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/definicao-de-politicas.png}
            \end{center}

            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item ISP só quer rotear tráfego de/para as redes de seus clientes (não quer carregar tráfego de trânsito entre outros ISPs – uma política típica do ``mundo real'')
                \item A anuncia o caminho Aw para B e para C
                \item B escolhe \textit{não} anunciar BAw para C!
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item B não recebe nenhuma ``receita'' por rotear CBAw, já que nenhum de C, A ou w é cliente de B
                    \item C não chega a aprender/conhecer o caminho CBAw
                \end{itemize} 
                \item x é \textbf{dual home} tem dois provedores de acesso: B e C
                \item \textbf{Política de ``Força'' (\textit{enforce}):} x não quer rotear trafego de B para C através de x, x não anuncia para B a rota para C 
            \end{itemize}

        \subsubsection*{Seleção das rotas BGP}

            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item o roteador pode aprender mais de uma rota para um AS de destino e seleciona a rota com base em:
                \begin{enumerate}[left=0.5cm, align=left, nosep]
                    \item Atributo de valor de preferência local (\textit{local preference}): decisão de política
                    \item Caminho de AS mais curto (\textit{shortest AS-PATH})
                    \item Roteador PRÓXIMO-SALTO (\textit{NEXT-HOP}) mais próximo: roteamento de ``batata quente'' (\textit{hot potato routing})
                    \item Critérios adicionais
                \end{enumerate}
            \end{itemize}

    \subsection{ICMP: Internet Control Message Protocol}
        
        \begin{itemize}[left=0.5cm, align=left, nosep]
            \item Usado por roteadores e sistemas finais para comunicar informações do nível da camada de rede 
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Reportar erros: uma máquina hospedeira, rede, porta, protocolo não foi alcançável, congestionamento, tempo de vida excedido, etc.
                \item Fazer requisições e respostas de echo (ping)   
            \end{itemize}
            \item Troca de pacotes ICMP é feita acima do IP
            \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                \item Cada mensagem ICMP é encapsulada/transportada em um datagrama IP   
            \end{itemize}
            \item Mensagem ICMP: Tipo, código, checksum, dados e os primeiros 8 bytes do datagrama IP que causou o erro
        \end{itemize}

        \subsubsection*{Traceroute e ICMP}

            \begin{center}
                \includegraphics[width=0.5\textwidth]{img/cap-04.01/traceroute-icmp.png}
            \end{center}

            \begin{itemize}[left=0.5cm, align=left, nosep]
                \item A origem envia conjuntos de segmentos UDP para o destino
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item O 1$^\text{o}$ conjunto tem TTL=1, o 2$^\text{o}$ conjunto tem TTL=2, etc.
                \end{itemize}

                \item O datagrama do $n$-ésimo conjunto chega ao $n$-ésimo roteador:
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item O roteador descarta o datagrama e envia para a origem uma mensagem ICMP (tipo 11, código 0)
                    \item A mensagem ICMP possivelmente inclui o nome do roteador e seu endereço IP
                \end{itemize}

                \item Quando a mensagem ICMP chega à origem: registrar os RTTs

                \item \textbf{Critérios de parada:}
                \begin{itemize}[left=0.5cm, nosep, label=$\hookrightarrow$]
                    \item O segmento UDP eventualmente chega ao host de destino
                    \item O destino retorna uma mensagem ICMP ``porta inacessível'' (tipo 3, código 3)
                    \item A origem interrompe o envio
                \end{itemize}
            \end{itemize}
